// automation.rs — top-level check invocation pipeline
pub fn invoke_check(ctx: &CheckContext, check: CheckRef) -> InvocationOutcome {
    // 1. dedup gate: consult pending-check set identity + freshness window
    if dedup::is_duplicate(ctx, &check) {
        return InvocationOutcome::Suppressed;
    }

    // 2. state mutation: mark the check as running on this invocation
    let token = ctx.state.begin(check.id());

    // 3. demotion check: if the check has been demoted, handle accordingly
    if demote::is_demoted(ctx, &check) {
        // demoted checks are not re-run; they are recorded and skipped
        ctx.state.record_skipped(token);
        return InvocationOutcome::SkippedDemoted;
    }

    // 4. execute
    let result = check.execute(ctx);

    // 5. on failure, demote
    if result.is_err() {
        demote::demote(ctx, &check);
    }

    // 6. end state
    ctx.state.end(token, &result);

    result
}

pub fn is_duplicate(ctx: &CheckContext, check: &CheckRef) -> bool {
    // A duplicate is: same check type + same pending-check set
